=== *args — 可变位置参数 ===
  args = (1, 2, 3), type = tuple
sum_all(1,2,3) = 6
  args = (1, 2, 3, 4, 5), type = tuple
sum_all(1,2,3,4,5) = 15
  args = (1, 2, 3, 4, 5), type = tuple
sum_all(*nums) = 15
